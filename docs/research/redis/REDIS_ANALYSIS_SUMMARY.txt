================================================================================
ROOT CAUSE ANALYSIS
================================================================================

ISSUE:
All 5 tests attempt to create RedisPool without timeout protection:

    let redis_url = std::env::var("REDIS_URL")
        .unwrap_or_else(|_| "redis://127.0.0.1:6379".to_string());
    let pool = match RedisPool::new(&redis_url).await {  // <-- HANGS HERE
        Ok(p) => p,
        Err(_) => {
            println!("SKIPPED: Redis not available");
            return;
        }
    };

PROBLEM:
- RedisPool::new() calls ConnectionManager::new() which attempts TCP connection
- When Redis is unavailable, TCP socket waits for system timeout (~60 seconds)
- No tokio::time::timeout() wrapper = no early escape
- Result: Each test blocks for 60+ seconds
- Total: 5 tests × 60 seconds = 300+ second suite hang

EVIDENCE:
From src/queue/redis_client.rs (lines 19-22):

    pub async fn new(redis_url: &str) -> Result<Self, RedisError> {
        let client = Client::open(redis_url)?;
        let conn_manager = ConnectionManager::new(client.clone()).await?;  // BLOCKS
        Ok(Self { conn_manager, client })
    }

    async fn get_test_redis() -> Option<RedisPool> {
        let redis_url = std::env::var("REDIS_URL")
            .unwrap_or_else(|_| "redis://127.0.0.1:6379".to_string());

        // Timeout 1: 5-second limit on connection
        match tokio::time::timeout(
            tokio::time::Duration::from_secs(5),
            RedisPool::new(&redis_url),
        )
        .await
        {
            Ok(Ok(pool)) =>
                // Timeout 2: 5-second health check
                match tokio::time::timeout(
                    tokio::time::Duration::from_secs(5),
                    pool.health_check(),
                )
                .await
                {
                    Ok(Ok(())) => Some(pool),
                    _ => None,
                }
            _ => None,
        }
    }

    macro_rules! setup_redis_or_skip {
        () => {{
            match get_test_redis().await {
                Some(pool) => pool,
                None => {
                    println!("SKIPPED: Redis not available");
                    return;
                }
            }
        }};
    }
